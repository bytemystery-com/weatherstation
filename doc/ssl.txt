with CA

mkdir -p ~/weatherd-ca
cd ~/weatherd-ca
chmod 700 ~/weatherd-ca

openssl genrsa -aes256 -out weatherd-ca.key 4096
chmod 600 weatherd-ca.key

openssl req -x509 -new -sha256 -key weatherd-ca.key -out weatherd-ca.crt -days 3650 \
  -subj "/CN=repro CA" \
  -addext "basicConstraints=critical,CA:TRUE" \
  -addext "keyUsage=critical,keyCertSign,cRLSign"

mkdir -p ~/weatherd-cert
cd ~/weatherd-cert
chmod 700 ~/weatherd-cert

openssl genrsa -out weatherd.key 3072
chmod 600 weatherd.key
openssl req -new \
  -key weatherd.key \
  -out ../weatherd-ca/weatherd.csr \
  -subj "/CN=weather-station"

cd ~/weatherd-ca
cat > weatherd-ext.cnf <<'EOF'
basicConstraints = critical,CA:FALSE
keyUsage = critical,digitalSignature,keyEncipherment
extendedKeyUsage = serverAuth
subjectAltName = @alt_names

[alt_names]
DNS.1 = weather-station
DNS.2 = weather-station.local
IP.1 = 192.168.0.130
IP.2 = 192.168.9.183
EOF

vim weatherd-ext.cnf
